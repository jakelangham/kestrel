%%ListType
% Dam break with Coulomb friction on a uniform slope (Mangeney et al. 2000):
% a reservoir 10 m deep (normal to the bed) on -1000 m < x < 0, released at
% t = 0 onto the dry slope b = -0.5 x, with friction coefficient 0.3, so that
% the flow accelerates uniformly. With geometric factors on, the solution is
% that of Ritter with gravity g cos(theta)^3 in a frame moving with
% acceleration g (s - mu) cos(theta)^2, where tan(theta) = s (see benchlib.jl). The initial condition is
% written by make_dambreak_ic in benchlib.jl, and must not be replaced by a Cap
% block. The errors are measured where the far end of the reservoir has no
% effect yet (see dambreak_errors).
Domain:
Lat = 0
Lon = 0
nXtiles = 50
nYtiles = 1
nXpertile = 10
nYpertile = 1
Xtilesize = 100
Boundary Conditions = halt

Parameters:
Drag = Coulomb
Coulomb Co = 0.3
Geometric factors = on
Erosion = Off
Deposition = None
Eddy Viscosity = 0

Solver:
T start = 0
T end = 30
Initial condition = dambreak_coulomb_slope_ic/000000
cfl = 0.5
max dt = 1
Tile Buffer = 3
Height threshold = 1e-6

Output:
N out = 3
directory = dambreak_coulomb_slope/

Topog:
Type = Function
Topog function = xslope
Topog params = (-0.5)
